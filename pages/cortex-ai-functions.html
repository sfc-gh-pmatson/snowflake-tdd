SELECT
  support_ticket_id,
  AI_CLASSIFY(description,
    ['billing', 'technical', 'account']
  ) AS category,
  AI_SENTIMENT(description) AS sentiment,
  AI_EXTRACT(description,
    'product_name'
  ) AS product
FROM support_tickets
WHERE AI_FILTER(
  description,
  'mentions a deadline or SLA'
);
