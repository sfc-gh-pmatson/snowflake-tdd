import streamlit as st
from snowflake.snowpark.context import get_active_session

session = get_active_session()

st.title("Sales Dashboard")
region = st.selectbox("Region", ["Americas", "EMEA", "APAC"])

df = session.sql(f"SELECT * FROM sales WHERE region = '{region}'")
st.line_chart(df.to_pandas(), x="DATE", y="REVENUE")
