-- Step 1: Create an Interactive Warehouse (always-on, pre-warmed cache)
CREATE OR REPLACE INTERACTIVE WAREHOUSE my_interactive_wh
  AUTO_SUSPEND  = 86400   -- min 24 hours; warehouse stays warm
  AUTO_RESUME   = TRUE
  INITIALLY_SUSPENDED = FALSE;

-- Step 2: Create an Interactive Table (requires CLUSTER BY)
CREATE OR REPLACE INTERACTIVE TABLE my_db.my_schema.events (
  event_id   BIGINT
, event_ts   TIMESTAMP_NTZ
, user_id    STRING
, event_type STRING
, payload    VARIANT
)
  CLUSTER BY (user_id, event_ts)  -- required; choose cols matching your WHERE clauses
  DATA_RETENTION_TIME_IN_DAYS = 1;

-- Step 3: Ingest via Snowpipe Streaming (SDK insertRows)
-- or load from existing standard table
INSERT INTO events
  SELECT * FROM staging.raw_events;

-- Step 4: Query using the Interactive Warehouse
USE WAREHOUSE my_interactive_wh;
SELECT event_type, COUNT(*) AS cnt
  FROM   events
  WHERE  user_id = 'u_12345'
    AND  event_ts >= DATEADD('hour', -1, CURRENT_TIMESTAMP())
  GROUP BY 1;

Interactive Warehouse

An always-on warehouse with a pre-warmed cache. The minimum AUTO_SUSPEND is 24 hours — you pay for uptime, but sub-second queries are guaranteed from the first request.

Interactive Table

A new table type that pairs with the Interactive Warehouse. The required CLUSTER BY clause tells Snowflake which columns to index for low-latency lookups — choose columns that appear in your most critical WHERE filters.

Ingestion

Load data via standard INSERT/COPY or stream rows in real time using the Snowpipe Streaming SDK (insertRows). The Interactive Warehouse serves the data immediately after ingestion.
