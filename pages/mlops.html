# End-to-end: train, log, deploy, predict
from snowflake.ml.registry import Registry
from snowflake.ml.modeling.xgboost import XGBClassifier

# Train on Snowflake compute
model = XGBClassifier(input_cols=features, label_cols=["CHURNED"])
model.fit(train_df)

# Log to Model Registry with metrics
reg = Registry(session)
mv = reg.log_model(
    model,
    model_name="churn_xgb",
    version_name="v1",
    metrics={"auc": 0.94}
)

# Serve — call from SQL or Python
mv.run(test_df, function_name="predict")
