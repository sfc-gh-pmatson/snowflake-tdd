-- 1. Create a Cortex Search Service
CREATE OR REPLACE CORTEX SEARCH SERVICE support_kb_search
  ON body
  ATTRIBUTES title, category, created_at
  WAREHOUSE = compute_wh
  TARGET_LAG = '1 hour'
  AS (
    SELECT body, title, category, created_at
    FROM support.articles
  );

-- 2. Query directly via SQL
SELECT r.value
FROM (
  SELECT SNOWFLAKE.CORTEX.SEARCH_PREVIEW(
    'support_kb_search',
    '{"query": "billing invoice error", "columns": ["title","body"], "limit": 5}'
  ) AS results
), LATERAL FLATTEN(results:results) r;

-- 3. Use as an Agent tool
CREATE OR REPLACE AGENT support_agent
  MODEL = 'claude-sonnet-5'
  TOOLS = (support_kb_search);

Cortex Search

Define the source table and column to search on. Snowflake handles all embedding generation, indexing, and serving automatically.

ATTRIBUTES are additional columns returned with results for filtering and display.

TARGET_LAG keeps the index in sync as new rows land — no manual refresh needed.

Add the service as a tool to any Cortex Agent and it will search autonomously at reasoning time.
